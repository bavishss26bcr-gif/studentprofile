/* SAFE HTML OUTPUT */

function escapeHTML(value = "") {
    return String(value).replace(/[&<>"']/g, char => ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#039;"
    })[char]);
}

function initials(name) {
    return name.trim().split(/\s+/).slice(0, 2)
        .map(word => word[0]).join("").toUpperCase();
}

/* RENDER DASHBOARD */

function renderDashboard() {

    const total = students.length;

    const active = students.filter(
        s => s.status === "Active"
    ).length;

    const cgpa = total
        ? students.reduce((sum, s) => sum + Number(s.cgpa), 0) / total
        : 0;

    const attendance = total
        ? students.reduce((sum, s) => sum + Number(s.attendance), 0) / total
        : 0;

    document.getElementById("totalStudents").textContent = total;

    document.getElementById("activeStudents").textContent = active;

    document.getElementById("averageCgpa").textContent = cgpa.toFixed(2);

    document.getElementById("averageAttendance").textContent =
        Math.round(attendance) + "%";

    renderStudents();
    renderDepartments();
}

/* DISPLAY STUDENTS */

function renderStudents() {

    const search = document.getElementById("search")
        .value.toLowerCase().trim();

    const department = document.getElementById("filter").value;

    const filtered = students.filter(student => {

        const matchesSearch =
            `${student.name} ${student.roll} ${student.email}`
            .toLowerCase().includes(search);

        const matchesDepartment =
            !department || student.department === department;

        return matchesSearch && matchesDepartment;
    });

    const table = document.getElementById("studentTable");

    table.innerHTML = filtered.map(student => `

        <tr>

            <td>
                <span class="avatar">
                    ${escapeHTML(initials(student.name))}
                </span>
                ${escapeHTML(student.name)}
                <br>
                <small>${escapeHTML(student.roll)}</small>
            </td>

            <td>${escapeHTML(student.department)}</td>

            <td>${student.year}</td>

            <td>${Number(student.cgpa).toFixed(2)}</td>

            <td>${student.attendance}%</td>

            <td>
                <span class="status ${student.status === "Inactive" ? "inactive" : ""}">
                    ${escapeHTML(student.status)}
                </span>
            </td>

            <td>
                <button class="action-btn"
                    onclick="viewStudent(${student.id})">
                    View
                </button>

                <button class="action-btn"
                    onclick="editStudent(${student.id})">
                    Edit
                </button>

                <button class="action-btn delete-btn"
                    onclick="deleteStudent(${student.id})">
                    Delete
                </button>
            </td>

        </tr>

    `).join("");

    document.getElementById("emptyMessage").style.display =
        filtered.length ? "none" : "block";
}

/* DEPARTMENT STATISTICS */

function renderDepartments() {

    const departments = {};

    students.forEach(student => {
        departments[student.department] =
            (departments[student.department] || 0) + 1;
    });

    const list = document.getElementById("departmentList");

    const entries = Object.entries(departments)
        .sort((a, b) => b[1] - a[1]);

    const max = Math.max(1, ...entries.map(item => item[1]));

    list.innerHTML = entries.map(([name, count]) => `

        <div class="department">

            <p>
                ${escapeHTML(name)}
                <strong style="float:right">${count}</strong>
            </p>

            <div class="progress">
                <div style="width:${count / max * 100}%"></div>
            </div>

        </div>

    `).join("");
}

/* OPEN ADD FORM */

    document.getElementById("studentForm").reset();

    document.getElementById("modalTitle").textContent =
        "Add Student";

    document.getElementById("studentModal").classList.add("show");
}

/* CLOSE MODAL */

}

/* SAVE / UPDATE STUDENT */

document.getElementById("studentForm")
.addEventListener("submit", function(event) {

    event.preventDefault();

    const formData = new FormData(this);

    const student = Object.fromEntries(formData.entries());

    student.id = Number(document.getElementById("studentId").value)
        || Date.now();

    student.year = Number(student.year);
    student.semester = Number(student.semester);
    student.cgpa = Number(student.cgpa);
    student.attendance = Number(student.attendance);

    const duplicate = students.some(s =>
        s.roll.toLowerCase() === student.roll.toLowerCase()
        && s.id !== student.id
    );

    if (duplicate) {
        alert("This roll number already exists!");
        return;
    }

    const index = students.findIndex(s => s.id === student.id);

    if (index >= 0) {
        students[index] = student;
        showToast("Student updated successfully!");
    } else {
        students.unshift(student);
        showToast("Student added successfully!");
    }

    saveData();

    renderDashboard();

    closeModal("studentModal");

});

/* EDIT STUDENT */

function editStudent(id) {

    const student = students.find(s => s.id === id);

    if (!student) return;

    openForm();

    document.getElementById("modalTitle").textContent =
        "Edit Student";

    document.getElementById("studentId").value = student.id;

    Object.keys(student).forEach(key => {

        const field = document.getElementById(key);

        if (field && key !== "id") {
            field.value = student[key];
        }

    });
}

                    <small>${label}</small>
                    <strong>${escapeHTML(value ?? "Not provided")}</strong>
                </div>

            `).join("")}

        </div>

        <br>

        <button class="add-btn"
            onclick="closeModal('profileModal');editStudent(${student.id})">
            Edit Profile
        </button>

    `;

    document.getElementById("profileModal").classList.add("show");
}

/* DELETE STUDENT */

function deleteStudent(id) {

    const student = students.find(s => s.id === id);

    if (!student) return;

    if (confirm("Are you sure you want to delete this student?")) {

        students = students.filter(s => s.id !== id);

        saveData();

        renderDashboard();

        showToast("Student deleted successfully!");

    }
}

/* EXPORT CSV */

function exportCSV() {

    if (!students.length) {
        showToast("No student records to export.");
        return;
    }

    const columns = [
        "name", "roll", "email", "phone", "dob",
        "gender", "department", "course", "year",
        "semester", "cgpa", "attendance", "status"
    ];

    const escapeCSV = value =>
        '"' + String(value ?? "").replace(/"/g, '""') + '"';

    const rows = [
        columns.join(","),
        ...students.map(student =>
            columns.map(key => escapeCSV(student[key])).join(",")
        )
    ];

    const blob = new Blob(
        ["\uFEFF" + rows.join("\n")],
        { type: "text/csv;charset=utf-8;" }
    );

    const url = URL.createObjectURL(blob);

    const link = document.createElement("a");

    link.href = url;
    link.download = "student-records.csv";

    link.click();

    URL.revokeObjectURL(url);

    showToast("Student report downloaded!");
}

/* NOTIFICATION */

let toastTimer;

function showToast(message) {

    const toast = document.getElementById("toast");

    toast.textContent = message;
    toast.style.display = "block";

    clearTimeout(toastTimer);

    toastTimer = setTimeout(() => {
        toast.style.display = "none";
    }, 2500);
}
